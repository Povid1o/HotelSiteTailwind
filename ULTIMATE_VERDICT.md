## ✅ Все необходимые исправления

### 1. Nginx конфигурация
```nginx
# Более специфичный regex
location ~ ^/static/(...)/.+\.(mp4|webm|avi|...)$ {
    # Range requests (критично!)
    proxy_set_header Range $http_range;
    proxy_set_header If-Range $http_if_range;
    proxy_cache_bypass $http_range;
    
    # CORS
    add_header 'Accept-Ranges' 'bytes' always;
    add_header 'Access-Control-Allow-Headers' 'Range, If-Range' always;
}
```

### 2. Backend MIME-типы
```javascript
app.use(express.static(..., {
  setHeaders: (res, filePath) => {
    if (filePath.endsWith('.mp4')) {
      res.setHeader('Content-Type', 'video/mp4');
    }
    res.setHeader('Accept-Ranges', 'bytes');
  }
}))
```

### 3. VideoPlayer - Нативный video
```typescript
const isLocalVideoFile = (url: string): boolean => {
    return !url.startsWith('http') || url.match(/\.(mp4|webm|...)$/i) !== null;
};

// Для локальных файлов
if (isLocal || isBlob) {
    return <video src={sourceUrl} controls crossOrigin="anonymous" ... />;
}

// Для YouTube
return <ReactPlayer url={sourceUrl} ... />;
```
